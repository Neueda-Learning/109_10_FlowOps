Table bank {
  bank_id bigint [pk, increment]
  bank_name varchar(100) [not null]
  swift_code varchar(20) [unique, not null]
  country_code char(3) [not null]
  is_active boolean
  created_at timestamp
}


Table currency {
  currency_code char(3) [pk]
  currency_name varchar(50) [not null]
  currency_symbol varchar(5)
  is_supported boolean
}


Table account {
  account_id bigint [pk, increment]

  bank_id bigint [not null]

  currency_code char(3) [not null]

  account_number varchar(50) [unique, not null]

  account_holder_name varchar(100) [not null]

  account_type varchar(20)
  // BUSINESS or PERSONAL

  country_code char(3)

  balance decimal(15,2)

  status varchar(20)
  // ACTIVE or BLOCKED

  created_at timestamp
}


Table receiver {

  receiver_id bigint [pk, increment]

  account_id bigint [unique, not null]

  employee_id varchar(20) [unique, not null]

  full_name varchar(100) [not null]

  email varchar(100)

  department varchar(100)

  employment_status varchar(20)

  created_at timestamp
}


Table payment {

  payment_id bigint [pk, increment]


  sender_account_id bigint [not null]


  receiver_id bigint [not null]


  amount decimal(15,2) [not null]


  currency_code char(3) [not null]


  payment_type varchar(20)


  status varchar(20)
  // CREATED
  // VALIDATED
  // SENT
  // COMPLETED
  // FAILED


  idempotency_key varchar(100) [unique, not null]


  reference varchar(255)


  error_code varchar(50)


  error_message varchar(255)


  scheduled boolean


  created_at timestamp


  updated_at timestamp
}



Table payment_status_history {

  history_id bigint [pk, increment]


  payment_id bigint [not null]


  old_status varchar(20)


  new_status varchar(20)


  remarks varchar(255)


  changed_at timestamp

}



Table scheduled_payment {

  schedule_id bigint [pk, increment]


  sender_account_id bigint [not null]


  receiver_id bigint [not null]


  amount decimal(15,2)


  currency_code char(3)


  frequency varchar(20)
  // WEEKLY
  // MONTHLY


  next_execution_date date


  last_execution_date date


  is_active boolean


  created_at timestamp
}



Table refund {

  refund_id bigint [pk, increment]


  payment_id bigint [not null]


  refund_amount decimal(15,2)


  reason varchar(255)


  refund_status varchar(20)
  // REQUESTED
  // APPROVED
  // COMPLETED
  // REJECTED


  requested_at timestamp


  completed_at timestamp
}



////////////////////////////////////////////////////
// RELATIONSHIPS
////////////////////////////////////////////////////


// Bank owns many accounts

Ref: bank.bank_id < account.bank_id


// Currency is used by many accounts

Ref: currency.currency_code < account.currency_code



// Employee receiver has one bank account

Ref: account.account_id - receiver.account_id



// Company account sends many payments

Ref: account.account_id < payment.sender_account_id



// Receiver receives many payments

Ref: receiver.receiver_id < payment.receiver_id



// Payment has many status history records

Ref: payment.payment_id < payment_status_history.payment_id



// Scheduled payments originate from company account

Ref: account.account_id < scheduled_payment.sender_account_id



// Scheduled payments target employees

Ref: receiver.receiver_id < scheduled_payment.receiver_id



// Scheduled payment currency

Ref: currency.currency_code < scheduled_payment.currency_code



// Payment can have refund records

Ref: payment.payment_id < refund.payment_id
